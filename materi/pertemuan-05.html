def attention(q,k,v,allowed=None):
    scores=q@k.T/np.sqrt(q.shape[-])
    =np.ones(scores.shape,dtype=bool) if allowed is None else allowed
    if mask.shape!=scores.shape or not np.all(mask.any(axis=-)):
        raise ValueError('Setiap query harus mempunyai minimal satu key yang diizinkan.')
    =np.where(mask,scores,-np.inf)
    exponent=np.exp(masked-np.max(masked,axis=-,keepdims=True))
    weights=exponent/exponent.sum(axis=-1,keepdims=True)
    return scores,weights,weights@v
